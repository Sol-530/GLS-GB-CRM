function getAlerts(){

  const out=[];

  leads().forEach(l=>(l.actions||[]).forEach(act=>{

    if(act.kind==='past')return;

    const diff=daysFrom(act.date);

    const u=act.done?'done':diff<0?'overdue':diff===0?'today':diff<=7?'upcoming':null;

    if(u)out.push({lead:l,action:act,urgency:u,diff:act.done?9999:diff});

  }));

  return out.sort((a,b)=>a.diff-b.diff);

}

function renderCalendar(){

  const y=S.calYear,m=S.calMonth;

  const MN=['January','February','March','April','May','June','July','August','September','October','November','December'];

  const firstDay=new Date(y,m,1).getDay();

  const daysInMonth=new Date(y,m+1,0).getDate();

  const daysInPrev=new Date(y,m,0).getDate();

  const tf=todayStr();

  const events={},allActs=[];

  leads().forEach(l=>(l.actions||[]).forEach(act=>{

    if(!act.date)return;

    const prefix=`${y}-${String(m+1).padStart(2,'0')}`;

    if(act.date.startsWith(prefix)){

      if(!events[act.date])events[act.date]=[];

      events[act.date].push({lead:l,action:act});

      allActs.push({lead:l,action:act,date:act.date});

    }

  }));

  allActs.sort((a,b)=>a.date<b.date?-1:1);

  let cells='',cc=0;

  for(let i=firstDay-1;i>=0;i--){cells+=`<div class="cal-cell"><div class="cal-day-num other-month">${daysInPrev-i}</div></div>`;cc++;}

  for(let d=1;d<=daysInMonth;d++){

    const ds=`${y}-${String(m+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;

    const isToday=ds===tf;

    const evs=events[ds]||[];

    const evHTML=evs.slice(0,3).map(function(item){

      const lead=item.lead,action=item.action;

      const cls=action.done?'ev-done':daysFrom(ds)<0?'ev-overdue':action.kind==='current'?'ev-current':'ev-next';

      return`<div class="cal-event ${cls}" title="${lead.name}: ${action.text}">${lead.name}</div>`;

    }).join('')+(evs.length>3?`<div style="font-size:10px;color:var(--text3);padding:1px 4px">+${evs.length-3} more</div>`:'');

    cells+=`<div class="cal-cell"><div class="cal-day-num${isToday?' today-num':''}">${d}</div>${evHTML}</div>`;cc++;

  }

  const rem=cc%7===0?0:7-(cc%7);

  for(let d=1;d<=rem;d++){cells+=`<div class="cal-cell"><div class="cal-day-num other-month">${d}</div></div>`;}

  const sideItems=allActs.length?allActs.map(function(item){

    const lead=item.lead,action=item.action,date=item.date;

    const diff=daysFrom(date);

    const color=action.done?'var(--text3)':diff<0?'var(--danger-text)':diff===0?'var(--warn-text)':'var(--info-text)';

    const bg=action.done?'var(--bg3)':diff<0?'var(--danger-bg)':diff===0?'var(--warn-bg)':'var(--info-bg)';

    return`<div class="act-sidebar-item">

      <div class="act-check ${action.done?'checked':''}" data-toggle="${lead.id}|${action.id}" style="margin-top:1px">

        ${action.done?'<i class="ti ti-check" style="font-size:10px;color:var(--bg)"></i>':''}

      </div>

      <div style="flex:1;min-width:0">

        <div style="font-size:12px;font-weight:500;color:${color};${action.done?'text-decoration:line-through':''}">${action.text}</div>

        <div style="font-size:11px;color:var(--text2);margin-top:1px">${lead.name} · ${date}</div>

        <div style="display:inline-block;font-size:10px;font-weight:500;padding:1px 6px;border-radius:20px;background:${bg};color:${color};margin-top:2px">${action.kind==='past'?'Past':action.kind==='current'?'Current':'Follow-up'}</div>

      </div>

    </div>`;

  }).join(''):`<div class="empty">No actions this month.</div>`;

  return`<div class="page-header">

    <div class="page-title">${MN[m]} ${y}</div>

    <div style="display:flex;gap:6px" class="no-print">

      <button class="btn btn-sm" id="cal-prev"><i class="ti ti-chevron-left"></i></button>

      <button class="btn btn-sm" id="cal-today">Today</button>

      <button class="btn btn-sm" id="cal-next"><i class="ti ti-chevron-right"></i></button>

    </div>

  </div>

  <div class="cal-layout">

    <div class="cal-wrap">

      <div class="cal-header"><div class="cal-title">${MN[m]} ${y}</div><div style="font-size:12px;color:var(--text3)">${allActs.length} action${allActs.length!==1?'s':''}</div></div>

      <div class="cal-grid">

        ${['Sun','Mon','Tue','Wed','Thu','Fri','Sat'].map(d=>`<div class="cal-dow">${d}</div>`).join('')}

        ${cells}

      </div>

    </div>

    <div class="act-sidebar">

      <div class="act-sidebar-head">Actions in ${MN[m]}</div>

      ${sideItems}

    </div>

  </div>`;

}

let importRows=[];

function processFile(file){

  const ext=file.name.split('.').pop().toLowerCase();

  if(ext==='csv'){const r=new FileReader();r.onload=e=>parseCSV(e.target.result);r.readAsText(file,'utf-8');}

  else if(ext==='xlsx'||ext==='xls'){const r=new FileReader();r.onload=e=>parseXLSX(e.target.result);r.readAsArrayBuffer(file);}

  else toast('Please upload .csv, .xlsx, or .xls');

}

function parseCSV(text){

  const lines=text.trim().split(/\r?\n/);

  if(lines.length<2){toast('File is empty.');return;}

  const headers=lines[0].split(',').map(h=>h.trim().replace(/^"|"$/g,'').toLowerCase());

  const rows=lines.slice(1).map(line=>{

    const vals=[];let cur='',inq=false;

    for(let i=0;i<line.length;i++){if(line[i]==='"')inq=!inq;else if(line[i]===','&&!inq){vals.push(cur.trim());cur='';}else cur+=line[i];}

    vals.push(cur.trim());

    const obj={};headers.forEach((h,i)=>{obj[h]=(vals[i]||'').trim();});

    return obj;

  }).filter(r=>r.name);

  showPreview(rows);

}

function parseXLSX(buf){

  if(!window.XLSX){const s=document.createElement('script');s.src='https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js';s.onload=()=>doXLSX(buf);document.head.appendChild(s);}

  else doXLSX(buf);

}

function doXLSX(buf){

  try{

    const wb=XLSX.read(buf,{type:'arraybuffer'});

    const ws=wb.Sheets[wb.SheetNames[0]];

    const json=XLSX.utils.sheet_to_json(ws,{defval:''});

    const rows=json.map(r=>{const o={};Object.keys(r).forEach(k=>{o[k.toLowerCase().trim()]=String(r[k]).trim();});return o;}).filter(r=>r.name);

    showPreview(rows);

  }catch(e){toast('Failed to read Excel file.');}

}

async function autoLinkAccount(companyName){

  if(!companyName)return[];

  let match=accs().find(a=>a.name.trim().toLowerCase()===companyName.trim().toLowerCase());

  if(!match){

    const obj={name:companyName.trim(),type:'Company',pic:'',email:'',phone:'',address:'',comment:'Auto-created from lead',color:COLS[Math.floor(Math.random()*COLS.length)],leads:[]};

    const r=await addDoc(collection(db,'accounts'),obj);

    match={...obj,id:r.id};

    DB.accounts[r.id]=match;

    toast(`Account "${companyName}" auto-created.`);

  }

  return[match.id];

}
